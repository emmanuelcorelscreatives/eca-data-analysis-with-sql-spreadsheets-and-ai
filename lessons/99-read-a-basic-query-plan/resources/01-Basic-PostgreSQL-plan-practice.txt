EXPLAIN (FORMAT TEXT)
SELECT o.order_id
FROM (VALUES (103), (101), (102)) AS o(order_id)
ORDER BY o.order_id;

EXPLAIN (FORMAT TEXT)
SELECT o.order_id, c.segment
FROM (VALUES
    (101, 1),
    (102, 2),
    (103, 1)
) AS o(order_id, customer_id)
JOIN (VALUES
    (1, 'Retail'),
    (2, 'Wholesale')
) AS c(customer_id, segment)
    ON o.customer_id = c.customer_id
ORDER BY o.order_id;
