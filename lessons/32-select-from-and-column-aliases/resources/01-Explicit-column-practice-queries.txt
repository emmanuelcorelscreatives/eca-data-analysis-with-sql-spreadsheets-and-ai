    order_date,
    customer_id
FROM orders;

-- Readable output headings

    order_id AS "Order ID",
    order_date AS "Order Date",
    customer_id AS "Customer ID"
FROM orders;

-- Comparison only: all current columns
SELECT * FROM orders;
